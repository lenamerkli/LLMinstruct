"""MinerU2.5-Pro PDF-to-Markdown extraction using the `transformers` backend.

This module wraps `mineru-vl-utils` so a whole PDF can be turned into Markdown.
The model, processor and client are loaded lazily once and reused across calls.
"""

from functools import lru_cache
from pathlib import Path

from pdf2image import convert_from_path

MODEL_NAME = "opendatalab/MinerU2.5-Pro-2605-1.2B"
PDF_DPI = 144


@lru_cache(maxsize=1)
def _client():
    """Load the MinerU2.5-Pro model/processor/client once (singleton)."""
    from transformers import AutoProcessor, Qwen2VLForConditionalGeneration
    from mineru_vl_utils import MinerUClient

    model = Qwen2VLForConditionalGeneration.from_pretrained(
        MODEL_NAME, dtype="auto", device_map="auto"
    )
    processor = AutoProcessor.from_pretrained(MODEL_NAME, use_fast=True)
    return MinerUClient(
        backend="transformers",
        model=model,
        processor=processor,
        image_analysis=False,  # set True to enable image/chart analysis
    )


def pdf_to_markdown(
    pdf_path: str,
    max_pages: int | None = None,
    dpi: int = PDF_DPI,
) -> str:
    """Convert a PDF into Markdown via MinerU2.5-Pro.

    Args:
        pdf_path: Path to the PDF file.
        max_pages: Optional cap on the number of pages to process (None = all).
        dpi: Render resolution for page images fed to the model.

    Returns:
        The concatenated Markdown of all processed pages.
    """
    path = Path(pdf_path)
    if not path.exists():
        raise FileNotFoundError(f"PDF not found: {pdf_path}")

    from mineru_vl_utils.post_process import json2md

    images = convert_from_path(
        str(path),
        dpi=dpi,
        first_page=1,
        last_page=max_pages,
    )

    client = _client()
    pages: list[str] = []
    for index, image in enumerate(images, start=1):
        print(f"MinerU: extracting page {index}/{len(images)}")
        content_list = client.two_step_extract(image)
        pages.append(json2md(content_list))

    return "\n\n".join(pages)

Why is it just text + equations, but no markdown?
§a§
I'll inspect how MinerU's `json2md` and the two-step extract pipeline turn layout detections into Markdown.Next I’ll check the official `json2md` implementation and example outputs.Checking how the full MinerU pipeline builds Markdown vs. this lightweight `json2md`.**Short answer:** `json2md` does almost no Markdown formatting. It only concatenates each block’s `content` with blank lines. Layout types like `title` / `list` are ignored for syntax.

## Root cause

Your pipeline is:

1. `client.two_step_extract(image)` → list of `ContentBlock`s (`type`, `bbox`, `content`, …)
2. `json2md(content_list)` → string

The implementation of `json2md` is effectively:

```python
def json2md(json_data):
    content_list = []
    last_text_contd_idx = -1
    for bbox_info in json_data:
        ctype = bbox_info.get("type", "text")
        content = bbox_info["content"]
        if content:
            if bbox_info.get("merge_prev", False) and last_text_contd_idx >= 0:
                # optional space + merge into previous text
                ...
            else:
                content_list.append(content)
                if ctype == "text":
                    last_text_contd_idx = len(content_list) - 1
    return "\n\n".join(content_list)
```

So it:

- **does** join blocks with `\n\n`
- **does** merge truncated text via `merge_prev`
- **does not** turn `title` / `doc_title` / `paragraph_title` into `#` / `##`
- **does not** turn `list` / `list_item` into `-` / `1.` lists
- **does not** drop or special-case `page_number` (that’s why bare `1`, `2`, `3` show up)

Equations look “Markdown-ish” only because post-process wraps them:

```python
# in post_process
block.content = _add_equation_brackets(...)  # → \[ ... \]
```

Titles are recognized as plain strings (`"2.1 Gesamtenergie"`), not as `# 2.1 Gesamtenergie`. Any `•` / `-` you see came from the OCR/VLM text itself, not from type→Markdown mapping.

Layout detection *does* see structure (your debug log has `title`, `list`, `equation`, …). That structure is just not used when building the string.

`mineru-vl-utils` is a thin VLM wrapper. The full [MinerU](https://github.com/opendatalab/MinerU) toolkit does more end-to-end Markdown assembly; this helper does not.

## What you’re seeing is expected

| Detected type | What lands in MD today |
|---|---|
| `title` | plain text |
| `text` | plain text |
| `list` / items | plain text (bullets only if in the content) |
| `equation` | `\[ ... \]` |
| `page_number` | `"1"`, `"2"`, … |
| `table` | HTML (if present) |

So the result is “text + equations”, not structured Markdown.

## Fix: map types yourself

Replace or wrap `json2md` with something that uses `type`:

```python
from mineru_vl_utils.structs import BlockType

SKIP_TYPES = {
    BlockType.PAGE_NUMBER,
    BlockType.HEADER,
    BlockType.FOOTER,
    BlockType.PAGE_FOOTNOTE,
    BlockType.EQUATION_BLOCK,  # usually a container; content is in child equations
    BlockType.LIST,            # container; items are separate (if present)
}

TITLE_LEVEL = {
    BlockType.DOC_TITLE: 1,
    BlockType.TITLE: 2,
    BlockType.PARAGRAPH_TITLE: 3,
}

def blocks_to_markdown(blocks) -> str:
    parts: list[str] = []
    last_text_idx = -1

    for b in blocks:
        t = b.get("type") or b.type
        c = b.get("content") if isinstance(b, dict) else b.content
        if not c or not str(c).strip():
            continue
        if t in SKIP_TYPES:
            continue

        c = str(c).strip()

        # truncated paragraph merge (same idea as upstream json2md)
        merge_prev = bool(b.get("merge_prev", False)) if isinstance(b, dict) else False
        if merge_prev and last_text_idx >= 0 and t == BlockType.TEXT:
            prev = parts[last_text_idx]
            # CJK: no space; otherwise one space
            if any("\u4e00" <= ch <= "\u9fff" for ch in c):
                parts[last_text_idx] = prev + c
            else:
                parts[last_text_idx] = prev + " " + c
            continue

        if t in TITLE_LEVEL:
            parts.append("#" * TITLE_LEVEL[t] + " " + c.lstrip("#").strip())
            last_text_idx = -1
        elif t == BlockType.LIST_ITEM:
            line = c if c.startswith(("-", "*", "•", "·")) or c[:2].rstrip(".").isdigit() else f"- {c}"
            parts.append(line.replace("•", "-").replace("·", "-"))
            last_text_idx = -1
        elif t == BlockType.EQUATION:
            if not c.startswith("\\["):
                c = f"\\[\n{c}\n\\]"
            parts.append(c)
            last_text_idx = -1
        elif t == BlockType.TABLE:
            parts.append(c)  # already HTML from post_process
            last_text_idx = -1
        elif t in (BlockType.CODE, BlockType.ALGORITHM):
            parts.append(f"```\n{c}\n```")
            last_text_idx = -1
        else:
            parts.append(c)
            if t == BlockType.TEXT:
                last_text_idx = len(parts) - 1
            else:
                last_text_idx = -1

    return "\n\n".join(parts)
```

## Extra notes

1. **Lists:** Layout often emits a parent `list` plus line `text`s (or `list_item`s). Upstream post-process even does `list_item → text`. Bullets may be missing unless you add them from type or from geometry.
2. **Heading levels:** The model doesn’t always distinguish `doc_title` vs `paragraph_title`; mapping all `title` → `##` is usually fine for notes like yours.
3. **Full MinerU:** For production PDF→MD (TOC, better lists, paratext handling), the full `mineru` package is the intended path; `mineru-vl-utils` + `json2md` is deliberately minimal.
4. **Your sample** already has good equation extraction and readable text; it’s mainly missing structural Markdown markers.

**Bottom line:** Not a model failure — `json2md` never turns layout types into Markdown. Add a small type-aware formatter (as above) and you’ll get real headings, lists, and cleaner notes.

## Full MinerU is the better solution here

Your current code uses **`mineru-vl-utils` directly**, which exposes the lower-level VLM output. Full MinerU adds the missing document-processing stages:

- whole-document rather than isolated-page processing
- heading-level assignment
- Markdown rendering of titles as `#`, `##`, etc.
- page-header/footer/page-number removal from the main Markdown
- image and chart extraction
- HTML table rendering
- fenced code blocks
- `middle.json`, `content_list.json`, and `content_list_v2.json`
- cross-page post-processing

Its renderer explicitly converts title blocks into Markdown headings.

One limitation: Full MinerU preserves lists structurally in its JSON output, but the VLM Markdown renderer does not always synthesize perfect `-` markers. It primarily renders each detected list item on a separate line. Consequently, headings will improve substantially, while lists may still need minor post-processing.

## Drop-in Python wrapper

Because the documented interfaces are currently the CLI and FastAPI service, the safest Python integration is a small wrapper around the CLI rather than importing MinerU’s internal modules:

```python
"""Full MinerU PDF-to-Markdown wrapper."""

from pathlib import Path
import subprocess


def pdf_to_markdown(
    pdf_path: str,
    output_dir: str | None = None,
    max_pages: int | None = None,
    backend: str = "hybrid-engine",
    effort: str = "medium",
) -> str:
    """Convert a PDF to Markdown using the full MinerU pipeline.

    The output directory is retained because the generated Markdown may
    reference extracted images relative to that directory.
    """
    path = Path(pdf_path).expanduser().resolve()

    if not path.is_file():
        raise FileNotFoundError(f"PDF not found: {path}")

    if path.suffix.lower() != ".pdf":
        raise ValueError(f"Expected a PDF file: {path}")

    if max_pages is not None and max_pages < 1:
        raise ValueError("max_pages must be at least 1")

    if output_dir is None:
        output = path.parent / f"{path.stem}_mineru"
    else:
        output = Path(output_dir).expanduser().resolve()

    output.mkdir(parents=True, exist_ok=True)

    command = [
        "mineru",
        "-p",
        str(path),
        "-o",
        str(output),
        "-b",
        backend,
    ]

    if backend.startswith("hybrid"):
        command.extend(["--effort", effort])

    # MinerU's page indexes are zero-based and --end is inclusive.
    if max_pages is not None:
        command.extend(["--end", str(max_pages - 1)])

    subprocess.run(command, check=True)

    # The exact intermediate directory depends on the selected backend.
    markdown_files = list(output.rglob(f"{path.stem}.md"))

    if not markdown_files:
        raise RuntimeError(
            f"MinerU completed but no Markdown file was found under {output}"
        )

    if len(markdown_files) > 1:
        markdown_files.sort(
            key=lambda candidate: candidate.stat().st_mtime_ns,
            reverse=True,
        )

    markdown_path = markdown_files[0]
    return markdown_path.read_text(encoding="utf-8")
```

Usage:

```python

    effort="high",
)

```
