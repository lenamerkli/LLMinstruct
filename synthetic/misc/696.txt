Find or create an Angular 21 Component that renders code with highlighting.

I’d **create** a small wrapper component around official `highlight.js` rather than pull in an Angular-specific wrapper. The official docs support ES module imports, recommend importing only the languages you need to avoid loading everything, and expose `highlight()` / `highlightAuto()` APIs that return HTML strings. Highlight.js also recommends semantic `<pre><code>` markup, and Angular documents that template `innerHTML` bindings are sanitized. That makes this a clean fit for Angular’s template model without manual DOM mutation.

### 1) Install the dependency

```bash
npm install highlight.js
```

`highlight.js` documents npm installation, and the official repo also shows bundled theme files under `highlight.js/styles/...` such as `github.css`.

### 2) Add a theme

```css
/* src/styles.css */
@import 'highlight.js/styles/github.css';
```

### 3) Create the component

```ts
// code-highlight.component.ts
import {
  booleanAttribute,
  ChangeDetectionStrategy,
  Component,
  computed,
  input,
} from '@angular/core';
import hljs from 'highlight.js/lib/core';
import bash from 'highlight.js/lib/languages/bash';
import css from 'highlight.js/lib/languages/css';
import javascript from 'highlight.js/lib/languages/javascript';
import json from 'highlight.js/lib/languages/json';
import typescript from 'highlight.js/lib/languages/typescript';
import xml from 'highlight.js/lib/languages/xml';

const AUTO_DETECT_LANGUAGES = [
  'bash',
  'css',
  'javascript',
  'json',
  'typescript',
  'xml',
] as const;

hljs.registerLanguage('bash', bash);
hljs.registerLanguage('css', css);
hljs.registerLanguage('javascript', javascript);
hljs.registerLanguage('json', json);
hljs.registerLanguage('typescript', typescript);
hljs.registerLanguage('xml', xml);

@Component({
  selector: 'app-code-highlight',
  changeDetection: ChangeDetectionStrategy.OnPush,
  host: {
    class: 'code-highlight',
  },
  template: `
    <figure class="code-highlight__figure">
      @if (hasCaption()) {
        <figcaption class="code-highlight__caption" [id]="captionId()">
          {{ captionText() }}
        </figcaption>
      }

      <pre
        class="code-highlight__pre"
        tabindex="0"
        [attr.aria-labelledby]="hasCaption() ? captionId() : null"
        [attr.aria-label]="hasCaption() ? null : ariaLabelText()"
      >
        <code
          [class]="codeClasses()"
          [innerHTML]="highlightedHtml()"
        ></code>
      </pre>
    </figure>
  `,
  styleUrl: './code-highlight.component.css',
})
export class CodeHighlightComponent {
  private static nextId = 0;
  private readonly id = `code-highlight-${CodeHighlightComponent.nextId++}`;

  readonly code = input.required<string>();
  readonly language = input<string | null>(null);
  readonly caption = input('');
  readonly ariaLabel = input('Highlighted code example');
  readonly wrap = input(false, { transform: booleanAttribute });

  readonly captionText = computed(() => this.caption().trim());
  readonly ariaLabelText = computed(() => {
    const label = this.ariaLabel().trim();
    return label.length > 0 ? label : 'Highlighted code example';
  });
  readonly hasCaption = computed(() => this.captionText().length > 0);
  readonly captionId = computed(() => `${this.id}-caption`);
  readonly normalizedLanguage = computed(() => this.language()?.trim().toLowerCase() ?? '');

  readonly codeClasses = computed(() => {
    const classes = ['code-highlight__code', 'hljs'];
    const language = this.normalizedLanguage();

    if (this.wrap()) {
      classes.push('code-highlight__code--wrap');
    }

    if (language.length > 0) {
      classes.push(`language-${language}`);
    }

    return classes.join(' ');
  });

  readonly highlightedHtml = computed(() => {
    const source = this.code();
    const language = this.normalizedLanguage();

    if (language.length > 0 && hljs.getLanguage(language)) {
      return hljs.highlight(source, {
        language,
        ignoreIllegals: true,
      }).value;
    }

    return hljs.highlightAuto(source, [...AUTO_DETECT_LANGUAGES]).value;
  });
}
```

### 4) Use it

```ts
// demo.component.ts
import { ChangeDetectionStrategy, Component, signal } from '@angular/core';
import { CodeHighlightComponent } from './code-highlight.component';

@Component({
  selector: 'app-demo',
  imports: [CodeHighlightComponent],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <app-code-highlight
      caption="counter.component.ts"
      language="typescript"
      [code]="snippet()"
      wrap
    />
  `,
})
export class DemoComponent {
  readonly snippet = signal(`
import {ChangeDetectionStrategy, Component, signal} from '@angular/core';

@Component({
  selector: 'app-counter',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: \`
    <button type="button">{{ count() }}</button>
  \`,
})
export class CounterComponent {
  readonly count = signal(0);
}
`.trim());
}
```

A few notes:

- `highlight.js` supports `highlight()`, `highlightAuto()`, `getLanguage()`, and `registerLanguage()`, so you can safely fall back to auto-detection or add more grammars later. Its docs also note that importing only the languages you need avoids loading the full language set.
- Highlight.js lists **HTML** as an alias of the XML grammar, so `language="html"` works once `xml` is registered.
- This component should be structurally AXE-friendly: semantic `<figure>/<figcaption>/<pre><code>`, keyboard-focusable scroll region, and explicit labeling. **But** WCAG AA color contrast still depends on the theme you import, so validate or override any token colors that don’t meet your contrast target.
