def detect_model_company(model_name: str) -> str:
    """Detect the company name from a model name string."""

    # Clean the model name
    model_name = model_name.lower()
    for prefix in ['openrouter/', 'publicai/', ':free']:
        model_name = model_name.replace(prefix, '')

    if model_name.endswith('.gguf'):
        model_name = model_name.rsplit('-', 1)[0]

    # Define company detection rules
    COMPANY_RULES = [
        (['openai'], 'OpenAI'),
        (['deepseek'], 'DeepSeek'),
        (['ministral', 'mistral'], 'Mistral AI'),
        (['qwen', 'alibaba'], 'Alibaba'),
        (['gemma', 'gemini', 'google'], 'Google'),
        (['z-ai'], 'Z.ai'),
        (['microsoft'], 'Microsoft'),
        (['llama', 'meta'], 'Meta'),
        (['moonshotai'], 'Moonshot AI'),
        (['grok', 'x-ai'], 'X.com'),
        (['anthropic'], 'Anthropic'),
        (['nvidia'], 'NVIDIA'),
        (['swiss-ai', 'apertus'], 'EPFL and ETH Zurich'),
        (['ibm'], 'IBM'),
        (['baidu'], 'Baidu'),
        (['minimax'], 'MiniMax AI'),
        (['poolside'], 'Poolside AI'),
        (['cohere'], 'Cohere Inc.'),
    ]

    # Check each rule
    for prefixes, company in COMPANY_RULES:
        if any(model_name.startswith(prefix) for prefix in prefixes):
            return company

    raise NotImplementedError(f"Company detection not implemented for model: {model_name}")
