class MediaContent:
    def __init__(self, media_type: MediaType, hash_: t.Optional[str] = None, content: t.Optional[bytes] = None):
        self._media_type: MediaType = media_type
        self._hash = ''
        if hash_:
            self.hash_ = hash_
        elif content:
            self._save_content(content)
        else:
            raise ValueError('At least one of hash_ or content must be provided')

    def __str__(self) -> str:
        return self.__repr__()

    def __repr__(self) -> str:
        return f"MediaContent(media_type={self.media_type}, hash={self.hash})"

    def __dict__(self) -> dict[str, str]:
        return {'type': self.media_type, 'hash': self.hash}

    def _save_content(self, content) -> None:
        self._hash = sha256(content).hexdigest()
        kind = guess_file_type(content)
        if not kind:
            extension = 'bin'
        else:
            extension = kind.extension
        match self._media_type:
            case 'image':
                base_location = IMAGE_LOCATION
            case 'audio':
                base_location = AUDIO_LOCATION
            case 'video':
                base_location = VIDEO_LOCATION
            case 'other_media':
                base_location = OTHER_MEDIA_LOCATION
            case _:
                base_location = OTHER_MEDIA_LOCATION
        base_location.mkdir(parents=True, exist_ok=True)
        with open(base_location / f"{self._hash}.{extension}", 'wb') as f:
            f.write(content)
        return None

    @property
    def hash(self) -> str:
        return self._hash

    @hash.setter
    def hash(self, hash_: str) -> None:
        self._hash = hash_

    @property
    def media_type(self) -> MediaType:
        return t.cast(MediaType, self._media_type)

    @media_type.setter
    def media_type(self, media_type: MediaType) -> None:
        self._media_type = media_type

```python
from classes import *
from config import *
import typing as t
import pyarrow as pa
import pyarrow.parquet as pq
import json

FILE_SIZE_LIMIT = 4_000_000_000  # 4.0 GB (not GiB)

CHAT_SCHEMA = pa.schema([
    ('messages',     pa.string()),
    ('source',       pa.string()),
    ('collected_at', pa.string()),
    ('ai_enhanced',  pa.bool_()),
])

TEXT_SCHEMA = pa.schema([
    ('text',         pa.string()),
    ('source',       pa.string()),
    ('collected_at', pa.string()),
    ('ai_enhanced',  pa.bool_()),
])

# Maps a data directory Path -> (open ParquetWriter, path of the file it is writing to)
_active_writers: dict[Path, tuple[pq.ParquetWriter, Path]] = {}


def _next_shard_path(location: Path) -> Path:
    """Return the next unused shard path inside *location*."""
    location.mkdir(parents=True, exist_ok=True)
    existing = sorted(location.glob('*.parquet'))
    if not existing:
        return location / '00000.parquet'
    last_index = int(existing[-1].stem)
    return location / f'{last_index + 1:05d}.parquet'


def _get_writer(location: Path, schema: pa.Schema) -> pq.ParquetWriter:
    """
    Return the active ParquetWriter for *location*.

    If the current shard has reached FILE_SIZE_LIMIT the writer is closed
    and a fresh one is opened against the next shard path.
    """
    if location in _active_writers:
        writer, file_path = _active_writers[location]
        if file_path.stat().st_size < FILE_SIZE_LIMIT:
            return writer
        # Current shard is full – close it and fall through to open the next one
        writer.close()
        del _active_writers[location]

    file_path = _next_shard_path(location)
    writer = pq.ParquetWriter(str(file_path), schema, compression='snappy')
    _active_writers[location] = (writer, file_path)
    return writer


def _to_table(data: t.Union[ChatEntry, TextEntry]) -> tuple[pa.Table, pa.Schema]:
    """Serialise *data* to a single-row PyArrow Table."""
    d = data.__dict__()

    if isinstance(data, ChatEntry):
        return (
            pa.table(
                {
                    'messages':     [json.dumps(d['messages'])],
                    'source':       [d['source']],
                    'collected_at': [d['collected_at']],
                    'ai_enhanced':  [d['ai_enhanced']],
                },
                schema=CHAT_SCHEMA,
            ),
            CHAT_SCHEMA,
        )

    return (
        pa.table(
            {
                'text':         [d['text']],
                'source':       [d['source']],
                'collected_at': [d['collected_at']],
                'ai_enhanced':  [d['ai_enhanced']],
            },
            schema=TEXT_SCHEMA,
        ),
        TEXT_SCHEMA,
    )


def save(data: t.Union[ChatEntry, TextEntry]) -> None:
    if isinstance(data, ChatEntry):
        location = CHAT_ENTRY_LOCATION
    elif isinstance(data, TextEntry):
        location = TEXT_ENTRY_LOCATION
    else:
        raise ValueError('data must be either ChatEntry or TextEntry')

    location.mkdir(parents=True, exist_ok=True)

    table, schema = _to_table(data)
    writer = _get_writer(location, schema)
    writer.write_table(table)


def close_all() -> None:
    """Flush and close every open ParquetWriter. Call this on program exit."""
    for writer, _ in _active_writers.values():
        writer.close()
    _active_writers.clear()
```

---

### Key design decisions

| Decision | Rationale |
|---|---|
| **`pyarrow` + `ParquetWriter`** | Lets us append row-groups to a file incrementally without re-reading it, which keeps memory usage flat regardless of file size. |
| **Module-level `_active_writers` dict** | One long-lived writer per directory avoids the overhead of opening/closing a file on every `save()` call. |
| **Size check *before* writing** | After a write the file may slightly exceed the limit by at most one entry, which is negligible at the 4 GB scale. Checking before is simpler and avoids an extra `stat` call on the common (not-yet-full) path. |
| **Shard naming `00000.parquet`, `00001.parquet`, …`** | HuggingFace `datasets` auto-discovers this pattern and assigns correct split indices when you call `load_dataset`. |
| **`ChatEntry.messages` → `json.dumps`** | The nested `Conversation`/`Message` structure has no direct Parquet analogue; serialising to a JSON string keeps the schema flat and the file readable by any Parquet consumer, while still being fully round-trippable. |
| **`snappy` compression** | Good balance between compression ratio and (de)compression speed; natively supported by the HuggingFace datasets loader. |
| **`close_all()`** | Parquet files are not valid until the footer is written on close. Callers should register `close_all` with `atexit` or call it explicitly in a `finally` block. |

**Recommended usage:**

```python
import atexit
from writer import save, close_all

atexit.register(close_all)   # ensures clean shutdown even on exceptions

for entry in my_entries:
    save(entry)
```
